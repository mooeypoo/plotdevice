<Promise eyebrow="See the story" title="The shape of the book, at a glance." shot="story" alt="An outline on a board: Beginning, Middle and End, with cards linked to chapters and characters">

</Promise>

<Promise eyebrow="Keep them close" title="Your characters, beside the page." shot="references" alt="A chapter with the references shelf open, showing Bea Halmi's page pinned">

</Promise>
